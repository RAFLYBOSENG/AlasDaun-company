CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE IF NOT EXISTS admins (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  email text NOT NULL,
  password_hash text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT admins_email_not_empty CHECK (char_length(btrim(email)) BETWEEN 3 AND 254)
);

CREATE UNIQUE INDEX IF NOT EXISTS admins_email_lower_idx
  ON admins (lower(email));

CREATE TABLE IF NOT EXISTS menu_items (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL CHECK (char_length(btrim(name)) BETWEEN 1 AND 120),
  category text NOT NULL CHECK (category ~ '^[a-z0-9]+(-[a-z0-9]+)*$'),
  price integer NOT NULL CHECK (price > 0),
  description_id text NOT NULL CHECK (char_length(btrim(description_id)) BETWEEN 1 AND 1000),
  description_en text NOT NULL CHECK (char_length(btrim(description_en)) BETWEEN 1 AND 1000),
  image_url text NOT NULL CONSTRAINT menu_items_image_url_check
    CHECK (image_url ~ '^https?://' OR image_url ~ '^/asset/images/menu/[0-9a-f-]+\.(jpg|png|webp|gif)$'
      OR image_url ~ '^data:image/(jpeg|png|webp|gif);base64,[A-Za-z0-9+/]+=*$'),
  is_recommended boolean NOT NULL DEFAULT false,
  is_new boolean NOT NULL DEFAULT false,
  promo integer CHECK (promo IS NULL OR promo BETWEEN 1 AND 99),
  is_active boolean NOT NULL DEFAULT true,
  sort_order integer NOT NULL DEFAULT 0 CHECK (sort_order >= 0),
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS menu_items_public_order_idx
  ON menu_items (is_active, sort_order, name);

CREATE OR REPLACE FUNCTION set_menu_items_updated_at()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS menu_items_set_updated_at ON menu_items;

CREATE TRIGGER menu_items_set_updated_at
BEFORE UPDATE ON menu_items
FOR EACH ROW
EXECUTE FUNCTION set_menu_items_updated_at();

-- Bagian 2: Tabel dan data galeri
-- Alas Daun - Neon schema for editable gallery media
-- Run this file in the Neon SQL Editor after the existing schema.

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_type
    WHERE typname = 'gallery_category'
      AND typnamespace = 'public'::regnamespace
  ) THEN
    CREATE TYPE public.gallery_category AS ENUM (
      'momen_di_alas_daun',
      'tempat_di_alas_daun',
      'behind_the_scene'
    );
  END IF;
END
$$;

CREATE TABLE IF NOT EXISTS gallery_items (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  category public.gallery_category NOT NULL DEFAULT 'momen_di_alas_daun',
  media_type text NOT NULL CHECK (media_type IN ('image', 'video')),
  media_url text NOT NULL CHECK (
    media_url ~ '^https?://'
    OR media_url ~ '^data:(image|video)/(jpeg|png|webp|gif|mp4|webm|ogg);base64,[A-Za-z0-9+/]+=*$'
  ),
  alt_text text NOT NULL CHECK (char_length(btrim(alt_text)) BETWEEN 1 AND 160),
  sort_order integer NOT NULL DEFAULT 0 CHECK (sort_order >= 0),
  is_active boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS gallery_items_public_order_idx
  ON gallery_items (is_active, sort_order, created_at);

DROP INDEX IF EXISTS gallery_items_media_url_idx;

CREATE UNIQUE INDEX gallery_items_media_url_idx
  ON gallery_items (md5(media_url));

CREATE OR REPLACE FUNCTION set_gallery_items_updated_at()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS gallery_items_set_updated_at ON gallery_items;

CREATE TRIGGER gallery_items_set_updated_at
BEFORE UPDATE ON gallery_items
FOR EACH ROW
EXECUTE FUNCTION set_gallery_items_updated_at();

-- Bagian 3: Tabel, keyword moderasi, dan waktu update ulasan
-- Alas Daun - Neon schema for customer testimonials and moderation keywords.

CREATE TABLE IF NOT EXISTS testimonial_blocked_keywords (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  keyword text NOT NULL CHECK (char_length(btrim(keyword)) BETWEEN 2 AND 80),
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE UNIQUE INDEX IF NOT EXISTS testimonial_blocked_keywords_lower_idx
  ON testimonial_blocked_keywords (lower(keyword));

CREATE TABLE IF NOT EXISTS testimonials (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL CHECK (char_length(btrim(name)) BETWEEN 2 AND 80),
  rating integer NOT NULL CHECK (rating BETWEEN 1 AND 5),
  review_text text NOT NULL CHECK (char_length(btrim(review_text)) BETWEEN 10 AND 1000),
  status text NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected')),
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS testimonials_public_idx
  ON testimonials (status, created_at DESC);

CREATE OR REPLACE FUNCTION set_testimonials_updated_at()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS testimonials_set_updated_at ON testimonials;

CREATE TRIGGER testimonials_set_updated_at
BEFORE UPDATE ON testimonials
FOR EACH ROW
EXECUTE FUNCTION set_testimonials_updated_at();
